import math
import numpy as np


def control(obs: dict) -> dict[str, float]:
    dt = float(obs['dt'])
    idle = float(obs['idle_mw'])
    rated = float(obs['rated_mw'])
    p_raw = float(np.clip(obs['p_raw_mw'], idle, rated))
    p_prev = idle if obs['step'] == 0 else float(obs['p_prev_grid_mw'])
    soc = float(obs['soc'])
    cap = float(obs['battery_capacity_mwh'])
    p_max = float(obs['battery_max_power_mw'])

    dt_h = max(dt / 3600.0, 1e-9)
    p_dis_max = min(p_max, max(0.0, (soc - 0.105) * cap * 0.9 / dt_h))
    p_chg_max = min(p_max, max(0.0, (0.895 - soc) * cap / dt_h))

    max_ramp = (20.0 / 60.0) * dt * 0.98
    p_desired = float(np.clip(p_raw + 10.0 * (0.50 - soc), idle, rated))
    p_target = float(np.clip(p_desired, p_prev - max_ramp, p_prev + max_ramp))
    p_target = float(np.clip(p_target, idle, rated))

    diff = p_raw - p_target
    if diff > p_dis_max:
        p_batt = p_dis_max
        gpu_floor = idle
        gpu_cap = float(np.clip(p_target + p_dis_max, idle, rated))
    elif diff < -p_chg_max:
        p_batt = -p_chg_max
        gpu_floor = float(np.clip(p_target - p_chg_max, idle, rated))
        gpu_cap = rated
    else:
        p_batt = diff
        gpu_floor = idle
        gpu_cap = rated

    return {
        'gpu_floor_mw': float(gpu_floor),
        'gpu_cap_mw': float(gpu_cap),
        'battery_setpoint_mw': float(p_batt),
    }
